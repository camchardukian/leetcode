  var makeConnected = function (n, connections) {
  if (connections.length < n - 1) {
  return - 1;
  }

      const tree = Array.from({ length: n }, (_, i) => i);
      let operationsNeeded = n - 1;

      function getRoot(node) {
          while (node !== tree[node]) {
              node = tree[node];
          }
          return node;
      }

      function connect(node, node2) {
          let root = getRoot(node);
          let root2 = getRoot(node2);
          if (root == root2) {
              return;
          }
          tree[root] = root2;
          operationsNeeded--;
      }

      for (let i = 0; i < connections.length; i++) {
          connect(connections[i][0], connections[i][1]);
      }

      return operationsNeeded;

  };
