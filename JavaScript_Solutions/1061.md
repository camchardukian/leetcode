/**
 * @param {string} s1
 * @param {string} s2
 * @param {string} baseStr
 * @return {string}
 */
var smallestEquivalentString = function (s1, s2, baseStr) {
    const tree = Array.from({ length: 26 }, (_, i) => i);
    let result = '';

    function getRoot(node) {
        while (tree[node] !== node) {
            node = tree[node];
        }
        return node;
    }

    function connect(node, node2) {
        let root = getRoot(node);
        let root2 = getRoot(node2);

        if (root === root2) return;
        if (root < root2) {
            tree[root2] = tree[root];
        } else {
            tree[root] = tree[root2];
        }
    }

    for (let i = 0; i < s1.length; i++) {
        connect(s1[i].charCodeAt(0) - 97, s2[i].charCodeAt(0) - 97);
    }


    for (let i = 0; i < baseStr.length; i++) {
        result += String.fromCodePoint(getRoot(baseStr[i].charCodeAt(0) - 97) + 97)
    }

    return result;
};
// s1.length === s2.length.
// s1[i] and s2[i] are equivalent chars but not necessarily the same.
// s1 and s2 are also equivalent strings of baseStr.

// Our goal is to return the lexicographically smallest equivalent string of baseStr by using equivalency information from s1 and s2.

// a = b = c = d
