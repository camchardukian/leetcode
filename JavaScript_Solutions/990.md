/**
 * @param {string[]} equations
 * @return {boolean}
 */
var equationsPossible = function (equations) {
    const tree = Array.from({ length: 26 }, (_, i) => i);

    function getRoot(node) {
        let charNodeToInteger = node.charCodeAt(0) - 97;
        while (tree[charNodeToInteger] !== charNodeToInteger) {
            charNodeToInteger = tree[charNodeToInteger]
        }
        return charNodeToInteger
    }

    function connect(node, node2) {
        const root = getRoot(node);
        const root2 = getRoot(node2);

        if (root === root2) {
            return;
        }
        tree[root] = tree[root2];
    }

    for (let i = 0; i < equations.length; i++) {
        if (equations[i][1] === "=") {
            connect(equations[i][0], equations[i][3])
        }
    }
    for (let i = 0; i < equations.length; i++) {
        if (equations[i][1] === "!") {
            if (getRoot(equations[i][0]) === getRoot(equations[i][3])) {
                return false;
            }

        }
    }
    return true;
};

// equations = an array of strings

// equations[i] will equal "xi==yi" OR "xi!=yi"
// xy and yi are lowercase letters (can be same or different)
// xy and yi represent one letter variable names

// return true if we can assign integers to variable names to to satisfy all the given equations or false otherwise.
