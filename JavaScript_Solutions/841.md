## My Solution

```
/**
 * @param {number[][]} rooms
 * @return {boolean}
 */
var canVisitAllRooms = function (rooms) {
    const visited = new Set();

    function dfs(node) {
        if (visited.has(node)) return;
        visited.add(node);
        for (const neighbor of rooms[node]) {
            dfs(neighbor)
        }
    }
    dfs(0)
    return visited.size === rooms.length;
};
```
