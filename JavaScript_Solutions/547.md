## My Solution

```
/**
 * @param {number[][]} isConnected
 * @return {number}
 */
var findCircleNum = function (isConnected) {
    const tree = [...Array(isConnected.length)].map((_, i) => i);
    let provinceCount = tree.length;
    function getRoot(index) {
        while (index !== tree[index]) {
            index = tree[index];
        }
        return index;
    }

    function connect(index1, index2) {
        const root1 = getRoot(index1);
        const root2 = getRoot(index2);
        if (root1 === root2) return;
        tree[root1] = root2;
        provinceCount--;
    }

    for (let i = 0; i < tree.length; i++) {
        for (let j = i + 1; j < tree.length; j++) {
            if (isConnected[i][j] === 1) {
                connect(i, j);
            }
        }
    }
    return provinceCount;
};

// n cities
// some cities connected, some not.
// cities can be connected directly, or indirectly
// a province is a group of cities connected in any way
```
