## My Solution

```
/**
 * @param {number} n
 * @param {number[][]} edges
 * @param {number} source
 * @param {number} destination
 * @return {boolean}
 */
var validPath = function (n, edges, source, destination) {
    const tree = [...Array(n).keys()];

    for (let i = 0; i < edges.length; i++) {
        connect(edges[i]);
    }

    function getRoot(node) {
        while (node !== tree[node]) {
            node = tree[node];
        }
        return node;
    }

    function connect(nodes) {
        const root1 = getRoot(nodes[0]);
        const root2 = getRoot(nodes[1]);
        if (root1 === root2) return;
        tree[root2] = root1;
    }

    return getRoot(source) === getRoot(destination);
};
```
