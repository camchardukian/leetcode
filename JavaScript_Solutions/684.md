## My Solution

```
/**
 * @param {number[][]} edges
 * @return {number[]}
 */
var findRedundantConnection = function (edges) {

    const tree = Array.from({ length: edges.length + 1 }, (_, i) => i);
    const result = [];

    function getRoot(index) {
        while (tree[index] !== index) {
            index = tree[index];
        }
        return index;
    }

    function connect(index1, index2) {
        let root1 = getRoot(index1);
        let root2 = getRoot(index2);
        if (root1 === root2) {
            result.push(index1, index2);
            return;
        }
        tree[root1] = root2;
    }

    for (let i = 0; i < edges.length; i++) {
        if (result.length) {
            return result;
        }
        connect(edges[i][0], edges[i][1]);
    }
    return result;
};

// graph started as a tree with n nodes from 1 to n.

// added edge has two different vertices from 1 to n and was not an edge that already existed

// graph = array edges of length n

// edges[i] = [ai, bi] indicates there is an edge between ai and bi.
```
